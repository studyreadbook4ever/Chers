cmake_minimum_required(VERSION 3.20)
project(CHERS VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CHERS_SANITIZE "Enable address and undefined behavior sanitizers" "${CHRONOLANE_SANITIZE}")
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3)
if(CHERS_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
add_library(chers_core STATIC src/core.cpp)
target_include_directories(chers_core PUBLIC include)
add_library(chers_render STATIC src/render.cpp)
target_include_directories(chers_render PUBLIC include)
add_library(chers_ipc STATIC src/ipc.cpp)
target_include_directories(chers_ipc PUBLIC include)
target_link_libraries(chers_ipc PUBLIC Threads::Threads)
add_library(chers_client SHARED src/client.cpp src/ipc.cpp)
target_include_directories(chers_client PUBLIC include)
target_link_libraries(chers_client PRIVATE Threads::Threads)
set_target_properties(chers_client PROPERTIES VERSION 1.0.0 SOVERSION 1)
add_custom_command(TARGET chers_client POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E rm -f libchronolane_client.so libchronolane_client.so.1
  COMMAND ${CMAKE_COMMAND} -E create_symlink $<TARGET_FILE_NAME:chers_client> libchronolane_client.so
  COMMAND ${CMAKE_COMMAND} -E create_symlink $<TARGET_FILE_NAME:chers_client> libchronolane_client.so.1
  WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
add_executable(chers src/main.cpp)
target_link_libraries(chers PRIVATE chers_core chers_render chers_ipc PkgConfig::SDL3 Threads::Threads)
target_compile_definitions(chers PRIVATE
  CHERS_BUILD_TYPE="${CMAKE_BUILD_TYPE}"
  CHERS_SANITIZERS=$<BOOL:${CHERS_SANITIZE}>)
add_custom_command(TARGET chers POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E rm -f chronolane
  COMMAND ${CMAKE_COMMAND} -E create_symlink $<TARGET_FILE_NAME:chers> chronolane
  WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
add_executable(chers_client_example examples/client.c)
target_link_libraries(chers_client_example PRIVATE chers_client)
add_executable(pixel_agent examples/pixel_agent.cpp)
target_link_libraries(pixel_agent PRIVATE chers_client Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE chers_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(render_tests tests/render_tests.cpp)
  target_link_libraries(render_tests PRIVATE chers_render)
  add_test(NAME render COMMAND render_tests)
  add_executable(ipc_tests tests/ipc_tests.cpp)
  target_link_libraries(ipc_tests PRIVATE chers_ipc chers_client Threads::Threads)
  add_test(NAME ipc COMMAND ipc_tests)
  add_executable(pixel_agent_tests tests/pixel_agent_tests.cpp)
  target_link_libraries(pixel_agent_tests PRIVATE chers_client Threads::Threads)
  add_test(NAME pixel_agent COMMAND pixel_agent_tests)
  add_test(NAME verifier COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/verifier_tests.py)
  add_test(NAME codex_vision_bridge COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/codex_vision_bridge_tests.py)
  add_test(NAME codex_evaluation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/codex_evaluation_tests.py)
  add_test(NAME codex_report COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/codex_report_tests.py)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/main_cli_tests.py $<TARGET_FILE:chers>)
endif()
install(TARGETS chers chers_client RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
install(FILES include/chers/client.h DESTINATION include/chers)
install(FILES include/chronolane/client.h DESTINATION include/chronolane)
